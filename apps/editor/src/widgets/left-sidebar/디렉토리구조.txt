│   └── sub-panels/            # 5개 서브 패널 상세 구현

│       ├── WidgetPanel.tsx    # 정적 패널 (from constants)
│       ├── PagePanel.tsx      # 동적 패널 (from entities/page)
│       ├── SectionPanel.tsx   # 동적 패널 (from entities/section)
│       ├── ModalPanel.tsx     # 동적 패널 (from entities/modal)
│       │
│       └── base/          # 패널 제작용 공통 부품
│           ├── PanelBaseLayout.tsx # 공통 헤더 + 구분선 틀
│           ├── StaticContent.tsx   # 정적 리스트용 UI 규격
│           └── DynamicContent.tsx  # 동적 리스트용 UI 규격
│
└── index.ts               # Public API: 외부(Pages 등)에는 LeftSidebar만 노출
